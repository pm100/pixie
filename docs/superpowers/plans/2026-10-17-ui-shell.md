# UI Shell and Navigation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the one-window quick layout with task tabs (Browse, Map, Settings), a sectioned navigation (Library, Recent, When with month chips, busy-month bars and day badges), a Settings page that owns the photo folders, a viewer with a filmstrip and a pop-out window, and a selection bar.

**Architecture:** `App.svelte` stays the owner of library state (view, groups, selection, scan), so switching tabs never loses it; tabs only choose what is drawn. Settings is drawn as a page over the still-mounted Browse screen so the grid keeps its scroll position. New pure logic (tab list, date navigation model, filmstrip range, pop-out messages, viewer preference) lives in small tested `.ts` files. The backend gains a `since` date bound, folder statistics and folder removal. The pop-out is a second Tauri window that loads the same bundle with `?viewer=1` and talks to the main window with Tauri events.

**Tech Stack:** Tauri 2, Rust (rusqlite), Svelte 5 + TypeScript, vitest, svelte-check.

**Spec:** `docs/superpowers/specs/2026-10-17-ui-shell-design.md`

## Review Focus

1. Switching Browse → Settings → Browse (and Map ↔ Browse) with a selection and a scrolled grid: selection, scroll position and nav branch must still be there; the open viewer must not survive a tab switch behind the user's back.
2. Removing a photo folder while one of its photos is selected, open in the viewer or shown in the pop-out: nothing must show a stale or wrong photo; the library must refresh once.
3. The pop-out window: opened twice (must focus the existing one), closed with the OS close button (main window must notice and stop sending), opened with no selection, library refreshed behind it (it keeps its own snapshot unless following).
4. "Last 30 days" combined with kind filter, map bounds and the snapshot; undated items must never appear under Recent; the empty result must show a sensible message, not a blank grid.
5. A library with only undated items, with one year, with 20 years (decades), and in "Oldest first" order: the navigation must not crash, mis-order or hide anything.

```rust
    pub fn folder_infos(&self) -> Result<Vec<FolderInfo>> {
        let c = self.conn();
        let mut s = c
            .prepare(
                "SELECT f.id, f.path, COUNT(p.id) FROM folders f
                 LEFT JOIN photos p ON p.folder_id = f.id AND p.missing = 0
                 GROUP BY f.id ORDER BY f.path",
            )
            .map_err(db_err)?;
        let rows = s
            .query_map([], |r| Ok(FolderInfo { id: r.get(0)?, path: r.get(1)?, count: r.get(2)? }))
            .map_err(db_err)?
            .collect::<std::result::Result<Vec<_>, _>>()
            .map_err(db_err)?;
        Ok(rows)
    }

    /// Forget a folder and its items (the `ON DELETE CASCADE` on photos). Files are never touched.
    pub fn remove_folder(&self, id: FolderId) -> Result<()> {
        self.conn().execute("DELETE FROM folders WHERE id = ?1", params![id]).map_err(db_err)?;
        Ok(())
    }
```

```ts
export type TabId = 'browse' | 'map' | 'create' | 'share' | 'settings';
export interface TabInfo { id: TabId; label: string }

/** Create and Share are empty until their first feature ships; an empty tab is not shown. */
export function visibleTabs(features: { create: boolean; share: boolean }): TabInfo[] {
  const all: (TabInfo & { on: boolean })[] = [
    { id: 'browse', label: 'Browse', on: true },
    { id: 'map', label: 'Map', on: true },
    { id: 'create', label: 'Create', on: features.create },
    { id: 'share', label: 'Share', on: features.share },
    { id: 'settings', label: 'Settings', on: true },
  ];
  return all.filter((t) => t.on).map(({ id, label }) => ({ id, label }));
}
```

import { weekday } from './format';
import type { TreeNode } from './tree';

export type YearEntry =
  | { kind: 'year'; node: TreeNode }
  | { kind: 'decade'; key: string; label: string; count: number; firstGroup: number; lastGroup: number; years: TreeNode[] }
  | { kind: 'undated'; node: TreeNode };

/** Items per month of a year node (index 0 = January). Month nodes have keys `m:YYYY-MM`. */
export function monthCounts(year: TreeNode): number[] {
  const out = new Array<number>(12).fill(0);
  for (const m of year.children) out[Number(m.key.slice(-2)) - 1] = m.count;
  return out;
}

/** The busiest single month across the given years, so the bars of different years are comparable. */
export function maxMonthCount(years: TreeNode[]): number {
  let max = 1;
  for (const y of years) if (y.key !== 'undated') for (const c of monthCounts(y)) max = Math.max(max, c);
  return max;
}

/**
 * Rows for the "When" navigation: the last six years as years, older years folded into decades, Undated last.
 * The order of `roots` (newest or oldest first) is kept.
 */
export function yearEntries(roots: TreeNode[], thisYear: number): YearEntry[] {
  const out: YearEntry[] = [];
  let undated: TreeNode | undefined;
  for (const n of roots) {
    if (n.key === 'undated') { undated = n; continue; }
    const year = Number(n.label);
    if (year >= thisYear - 5) { out.push({ kind: 'year', node: n }); continue; }
    const decade = Math.floor(year / 10) * 10;
    const key = `decade:${decade}`;
    const last = out[out.length - 1];
    if (last && last.kind === 'decade' && last.key === key) {
      last.years.push(n);
      last.count += n.count;
      last.firstGroup = Math.min(last.firstGroup, n.firstGroup);
      last.lastGroup = Math.max(last.lastGroup, n.lastGroup);
    } else {
      out.push({ kind: 'decade', key, label: `${decade}s`, count: n.count, firstGroup: n.firstGroup, lastGroup: n.lastGroup, years: [n] });
    }
  }
  if (undated) out.push({ kind: 'undated', node: undated });
  return out;
}

/** The pieces of a day row: day of month, short weekday and whether it falls on a weekend. */
export function dayBadge(day: string): { dom: number; weekday: string; weekend: boolean } {
  const w = weekday(day);
  return { dom: Number(day.slice(8, 10)), weekday: w, weekend: w === 'Sat' || w === 'Sun' };
}

import type { View } from './api';
import { parseKindFilter } from './kind';

export const POPOUT_SHOW = 'viewer-show';
export const POPOUT_READY = 'viewer-ready';
export const POPOUT_MOVED = 'viewer-moved';
export const POPOUT_CLOSED = 'viewer-closed';

/** `follow` is true for messages sent because the main window's selection changed, false for an explicit "open this". */
export interface ShowMessage { index: number; total: number; view: View; follow: boolean }

const isInt = (v: unknown): v is number => typeof v === 'number' && Number.isInteger(v);

function parseView(v: unknown): View | null {
  if (typeof v !== 'object' || v === null) return null;
  const o = v as Record<string, unknown>;
  if (o.order !== 'newest' && o.order !== 'oldest') return null;
  const num = (x: unknown) => (typeof x === 'number' ? x : null);
  return {
    order: o.order, folderId: num(o.folderId), kind: parseKindFilter(o.kind),
    bounds: (o.bounds ?? null) as View['bounds'], maxId: num(o.maxId), since: num(o.since),
  };
}

/** Events carry untrusted data; only a message that is fully valid is acted on. */
export function parseShow(v: unknown): ShowMessage | null {
  if (typeof v !== 'object' || v === null) return null;
  const o = v as Record<string, unknown>;
  if (!isInt(o.index) || !isInt(o.total) || o.total < 1 || o.index < 0 || o.index >= o.total) return null;
  const view = parseView(o.view);
  return view ? { index: o.index, total: o.total, view, follow: o.follow === true } : null;
}

export function parseMoved(v: unknown): { index: number } | null {
  if (typeof v !== 'object' || v === null) return null;
